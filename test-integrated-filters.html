🔧 Как интегрировать в существующий код:

  1. Добавить HTML-структуру фильтров в `interfaces/sales-interface.html`
  2. Добавить JavaScript логику в `scripts/sales-script.js`:
    • Функции для управления чипсами
    • Интеграция с существующим поиском ставок
    • Фильтрация результатов перед отображением
  3. Модифицировать функцию поиска чтобы учитывать выбранные фильтры
  4. Добавить CSS стили в `styles.css`

📝 Ключевые изменения в коде:

// 1. Добавить состояние фильтров
const activeFilters = {
    lines: [],
    agents: [],
    terminals: [],
    railAgents: []
};

// 2. Модифицировать функцию поиска
function searchWithFilters(baseResults) {
    return baseResults.filter(result => {
        // Проверка фильтра по линии
        if (activeFilters.lines.length > 0 &&
            !activeFilters.lines.some(line =>
                result.carrier && result.carrier.includes(line)
            )) {
            return false;
        }
        
        // Проверка фильтра по агенту
        if (activeFilters.agents.length > 0 &&
            !activeFilters.agents.some(agent =>
                result.agent && result.agent.includes(agent)
            )) {
            return false;
        }
        
        // Проверка фильтра по терминалу ЖД (для комплексных ставок)
        if (activeFilters.terminals.length > 0 &&
            result.transportType === 'sea_rail' &&
            !activeFilters.terminals.some(terminal =>
                result.data.rail && result.data.rail.agent &&
                result.data.rail.agent.includes(terminal)
            )) {
            return false;
        }
        
        return true;
    });
}

// 3. Интеграция с существующей функцией calculateAllRates
function calculateAllRatesWithFilters(departure, destination, containerType) {
    const baseResults = calculateAllRates(departure, destination, containerType);
    return searchWithFilters(baseResults);
}

🎯 Реализация чипсов:

// Управление чипсами
function addFilter(type, value) {
    if (!activeFilters[type].includes(value)) {
        activeFilters[type].push(value);
        updateChipsUI(type);
        updateActiveFiltersBar();
    }
}

function removeFilter(type, value) {
    activeFilters[type] = activeFilters[type].filter(item => item !== value);
    updateChipsUI(type);
    updateActiveFiltersBar();
}

function clearAllFilters() {
    Object.keys(activeFilters).forEach(key => {
        activeFilters[key] = [];
    });
    updateAllChipsUI();
    updateActiveFiltersBar();
}
